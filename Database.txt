-- =========================================
-- REPORTS
-- =========================================
CREATE TABLE reports(
    report_id INT AUTO_INCREMENT PRIMARY KEY,

    user_id INT NOT NULL,

    subdivision_id INT NOT NULL,

    category_id INT NOT NULL,

    pet_id INT NULL,

    animal_type ENUM(
        'Dog',
        'Cat',
        'Unknown'
    ) DEFAULT 'Unknown',

    animal_breed VARCHAR(100) NULL,

    animal_color VARCHAR(255) NULL,

    estimated_size ENUM(
        'Small',
        'Medium',
        'Large'
    ) NULL,

    description TEXT,

    latitude DECIMAL(10,8) NOT NULL,

    longitude DECIMAL(11,8) NOT NULL,

    animal_count INT DEFAULT 1,

    landmark VARCHAR(255),

    -- priority_level:
    --   - 'Low': Normal stray animal condition (no immediate threat or injuries).
    --   - 'Medium': Weak, sick, or scared stray animals (needs rescue/attention, not immediately dangerous).
    --   - 'High': Injured, aggressive, rabid, or attacking animals (requires immediate emergency response).
    priority_level ENUM(
        'Low',
        'Medium',
        'High'
    ) DEFAULT 'Medium',

    visibility ENUM(
        'Public',
        'Private'
    ) DEFAULT 'Public',

    is_possible_owned BOOLEAN DEFAULT FALSE,

    current_status_id INT,

    `condition` TEXT NULL,

    ai_animal_type VARCHAR(50) NULL,
    ai_dominant_color VARCHAR(100) NULL,
    ai_estimated_size VARCHAR(50) NULL,
    ai_possible_breed VARCHAR(100) NULL,
    ai_suggested_risk_level VARCHAR(50) NULL,
    ai_suggested_priority VARCHAR(50) NULL,
    ai_suggested_priority_reason TEXT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY(user_id)
    REFERENCES users(user_id)
    ON DELETE CASCADE,

    FOREIGN KEY(subdivision_id)
    REFERENCES subdivisions(subdivision_id),

    FOREIGN KEY(category_id)
    REFERENCES report_categories(category_id),

    FOREIGN KEY(pet_id)
    REFERENCES pets(pet_id)
    ON DELETE SET NULL,

    FOREIGN KEY(current_status_id)
    REFERENCES report_status(status_id)
    ON DELETE SET NULL
);

CREATE INDEX idx_reports_user
ON reports(user_id);

CREATE INDEX idx_reports_location
ON reports(latitude, longitude);

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY(announcement_id)
    REFERENCES announcements(announcement_id)
    ON DELETE CASCADE,

    FOREIGN KEY(user_id)
    REFERENCES users(user_id)
    ON DELETE CASCADE,

    FOREIGN KEY(parent_comment_id)
    REFERENCES announcement_comments(comment_id)
    ON DELETE CASCADE
);
